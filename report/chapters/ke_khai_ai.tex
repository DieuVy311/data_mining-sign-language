\begin{center}
  {\Large\bfseries Tuyên bố sử dụng công cụ AI (AI-Use Statement)}
\end{center}
\vspace{0.5em}

Theo quy định về liêm chính học thuật của học phần, nhóm kê khai việc sử dụng công cụ trí tuệ nhân tạo (AI) trong quá trình thực hiện bài tập lớn như sau.

\paragraph*{Công cụ sử dụng} Trợ lý AI Claude (Anthropic).

\paragraph*{Nội dung có sử dụng AI hỗ trợ}
\begin{itemize}
  \item Giải thích lý thuyết (SVM, PCA, LSTM, attention, các chỉ số đánh giá) và gợi ý hướng tiếp cận.
  \item Gợi ý cấu trúc mã nguồn cho các notebook tiền xử lý, trích xuất đặc trưng, SVM, CNN-LSTM và so sánh thuật toán; hỗ trợ gỡ lỗi khi chạy trên Kaggle và máy cá nhân.
  \item Gợi ý dàn ý một số mục của báo cáo.
\end{itemize}

\paragraph*{Phần nhóm tự thực hiện và kiểm chứng}
\begin{itemize}
  \item Chọn đề tài, xác định mục tiêu, quyết định hướng thực hiện, thuận toán, các tham số, cách đánh giá.
  \item Viết mã nguồn, chạy thí nghiệm, thu thập kết quả,
  \item Chạy toàn bộ thí nghiệm, nộp bài lên nền tảng AIChallenge, đối chiếu và kiểm tra lại các con số trong báo cáo với kết quả thực tế.
  \item Đọc, chỉnh sửa và chịu trách nhiệm về toàn bộ mã nguồn và nội dung báo cáo; các thành viên đều nắm các thuật toán đã dùng.
\end{itemize}
